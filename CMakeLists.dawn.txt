	LINK_DIRECTORIES( ${MY_INSTALL_DIR}/lib )


if(WIN32)
    set(DAWN_SHARED_BYPRODUCT "${MY_INSTALL_DIR}/lib/webgpu_dawn.lib") # Import lib for DLL
else()
    set(DAWN_SHARED_BYPRODUCT "${MY_INSTALL_DIR}/lib/libwebgpu_dawn.so") # Change to .dylib on Mac
endif()



if(WIN32)
    set( USE_CONFIGURATION "$<CONFIG>" )
    set( USE_BINDIR "<BINARY_DIR>_tmp/$<CONFIG>" )
    #set( USE_BINDIR "<BINARY_DIR>_tmp/\${CMAKE_INSTALL_CONFIG_NAME}" )
else()
    set( USE_CONFIGURATION "${CMAKE_BUILD_TYPE}" )
    set( USE_BINDIR <BINARY_DIR>_tmp/${CMAKE_BUILD_TYPE} )

	set( FORCE_CLANG
		-DCMAKE_C_COMPILER=clang
		-DCMAKE_CXX_COMPILER=clang++ )


endif()

# Which Dawn backends exist is a platform fact, not a preference, and it has to
# be stated explicitly in both directions.
#
# Dawn derives sensible per-platform defaults for these, but option() honours an
# explicitly passed -D cache value over its default — the same behaviour that
# makes TINT_BUILD_HLSL_WRITER below necessary. So passing D3D=ON unconditionally
# does not get politely ignored off-Windows: it forces Dawn to compile the D3D11
# and D3D12 backends against d3d12.h on a machine that has no such header.
if( WIN32 )
	set( DAWN_PLATFORM_BACKENDS
		-DDAWN_ENABLE_D3D11=ON         # Direct3D 11 backend
		-DDAWN_ENABLE_D3D12=ON         # Direct3D 12 backend — the XR interop path
		# Dawn defaults this to ${DAWN_ENABLE_D3D12}, but option() honours an
		# existing cache entry, so a tree first configured with D3D12 OFF keeps
		# the HLSL writer OFF forever and the D3D12 backend fails to build.
		# Pass it explicitly so a clean build (or another machine) works.
		-DTINT_BUILD_HLSL_WRITER=ON )
else()
	set( DAWN_PLATFORM_BACKENDS
		-DDAWN_ENABLE_D3D11=OFF
		-DDAWN_ENABLE_D3D12=OFF )
endif()

	ExternalProject_Add( Dawn_External
#    SOURCE_DIR      ${CMAKE_BINARY_DIR}/Dawn_External-prefix/src/Dawn_External
    GIT_REPOSITORY  "https://github.com/google/dawn"
    GIT_TAG         "v20260423.175430"
    GIT_SHALLOW     TRUE
    # ExternalProject runs `git submodule update --init --recursive` by default.
    # Dawn declares 61 submodules — Chromium infrastructure repos among them
    # (build, buildtools, testing, catapult, depot_tools) — and recursing into
    # those reaches repositories that require a Google-internal account, which
    # is where the chrome-internal credential prompt comes from.
    #
    # None of it is needed: DAWN_FETCH_DEPENDENCIES=ON below runs Dawn's own
    # tools/fetch_dawn_dependencies.py, which pulls the 19-dependency minimal
    # set a CMake build actually uses from public chromium.googlesource.com.
    # The submodules are redundant here, and enormous.
    #
    # An empty GIT_SUBMODULES means "none" only under policy CMP0097 (added in
    # 3.16); under OLD it means "all", i.e. the exact opposite. The top-level
    # CMakeLists sets 3.21 so this is NEW — but note that a
    # cmake_minimum_required() in an *included* file resets policies for the
    # rest of the scope, so keep those out of CMakeLists.gui.txt or this line
    # quietly stops working.
    GIT_SUBMODULES  ""
		BINARY_DIR ${CMAKE_BINARY_DIR}/Dawn
		INSTALL_DIR ${MY_INSTALL_DIR}
		BUILD_BYPRODUCTS "${DAWN_SHARED_BYPRODUCT}"
	CMAKE_ARGS -DCMAKE_INSTALL_PREFIX:PATH=${USE_BINDIR}
		${FORCE_CLANG}
		-DCMAKE_BUILD_TYPE:STRING=${USE_CONFIGURATION}
		-DCMAKE_CXX_FLAGS="-Wno-invalid-constexpr"
	        -DDAWN_ENABLE_INSTALL=ON       # <--- REQUIRED for Dawn to install files
        	-DDAWN_FETCH_DEPENDENCIES=ON   # <--- REQUIRED to auto-download Tint/Abseil deps
		#-DDAWN_SUPPORTS_DEPENDENCY_GTEST=OFF
	        -DDAWN_BUILD_MONOLITHIC_LIBRARY:STRING=SHARED
        	-DDAWN_BUILD_TESTS=OFF      # Disables Dawn's core testing suites
	        -DTINT_BUILD_TESTS=OFF      # Disables tests inside the Tint shader translator subproject
        	-DDAWN_BUILD_SAMPLES=OFF    # Disables example binaries (GLFW apps, sample code)
	        -DTINT_ENABLE_INSTALL=OFF   # Disables installing the Tint shader translator (not needed for runtime)
        	-DDAWN_ENABLE_VULKAN=ON        # Force compiles the Vulkan backend layer
	        ${DAWN_PLATFORM_BACKENDS}      # D3D11/D3D12 + HLSL writer on Windows; off elsewhere
	        # Dawn builds a C++20 module target (dawncpp_module) whenever its own
	        # probe in src/cmake/DawnCompilerChecks.cmake says the toolchain can
	        # handle modules. That probe accepts "Visual Studio" generators and
	        # accepts Clang as long as clang-scan-deps exists — and with clang-cl
	        # both are true, so it turns modules on. But the Visual Studio
	        # generator can only scan module dependencies with MSVC cl, not with
	        # the clang-cl toolset, so CMake's *generate* step then fails with
	        # "the compiler does not provide a way to discover the import graph".
	        # The probe is a check_cxx_source_compiles, which skips when the
	        # variable is already in the cache — so defining it here wins.
	        # We only ever #include <dawn/webgpu_cpp.h>, never `import dawn;`,
	        # so losing the module target costs nothing.
	        -DDAWN_SUPPORTS_CXX_MODULES=OFF
	        -DDAWN_ENABLE_METAL=OFF        # Disables Metal fallback (if compiling on Apple hardware)
        	-DDAWN_ENABLE_DESKTOP_GL=OFF   # Disables standard desktop OpenGL compatibility lines
	        -DDAWN_ENABLE_OPENGLES=OFF     # Disables mobile embedded OpenGLES targets

			${MOREOPTS}
	)
#else( INCLUDE_DAWN )
